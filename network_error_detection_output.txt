import zlib


def parity_bit(data):
    bits = "".join(format(byte, "08b") for byte in data)
    return "0" if bits.count("1") % 2 == 0 else "1"


def checksum(data):
    if len(data) % 2 != 0:
        data += b"\x00"

    total = 0
    for i in range(0, len(data), 2):
        word = (data[i] << 8) + data[i + 1]
        total += word
        total = (total & 0xFFFF) + (total >> 16)

    return (~total) & 0xFFFF


def crc_value(data):
    return zlib.crc32(data) & 0xFFFFFFFF


def main():
    # Input values used in the assignment output
    message = "HELLO NETWORK"
    total_frames = 10
    error_injection = True

    original_data = message.encode("utf-8")

    # Generate error-detection values at the sender
    parity = parity_bit(original_data)
    check_sum = checksum(original_data)
    crc = crc_value(original_data)

    successful_frames = 0
    detected_errors = 0
    receiver_check = "No error"

    for frame in range(1, total_frames + 1):
        received_data = original_data

        # Simulate a transmission error in the first frame
        if error_injection and frame == 1:
            changed_data = bytearray(original_data)
            changed_data[0] ^= 0b00000001
            received_data = bytes(changed_data)

        # Receiver verifies the CRC value
        received_crc = crc_value(received_data)

        if received_crc != crc:
            detected_errors += 1
            if frame == 1:
                receiver_check = "CRC mismatch"
        else:
            successful_frames += 1

    success_rate = (successful_frames / total_frames) * 100
    error_rate = (detected_errors / total_frames) * 100

    # Display the report
    RED = "\033[91m"
    GREEN = "\033[92m"
    YELLOW = "\033[93m"
    BLUE = "\033[94m"
    CYAN = "\033[96m"
    RESET = "\033[0m"

    print("\n" + "=" * 66)
    print("       Network Error Detection and LAN Performance Monitoring")
    print("=" * 66)

    print(f"{'Sender Device':24}: Device A")
    print(f"{'Receiver Device':24}: Device B")
    print(f"{'Input Data':24}: {message}")
    print(f"{'Parity Bit':24}: Generated ({parity})")
    print(f"{'Checksum':24}: Calculated ({check_sum:04X})")
    print(f"{'CRC':24}: Generated ({crc:08X})")

    print()
    print(f"{'Error Injection':24}: "
          f"{'Enabled' if error_injection else 'Disabled'}")

    check_color = YELLOW if detected_errors else GREEN
    status_color = RED if detected_errors else GREEN

    print(f"{'Receiver Check':24}: {check_color}{receiver_check}{RESET}")
    print(f"{'Error Detection':24}: "
          f"{status_color}{'ERROR DETECTED' if detected_errors else 'NO ERROR'}{RESET}")
    print(f"{'Transmission Status':24}: "
          f"{status_color}{'UNSUCCESSFUL' if detected_errors else 'SUCCESSFUL'}{RESET}")

    print()
    print(f"{CYAN}PERFORMANCE REPORT{RESET}")
    print(f"{'Total Frames':24}: {total_frames}")
    print(f"{GREEN}{'Successful Frames':24}: {successful_frames}{RESET}")
    print(f"{YELLOW}{'Detected Error':24}: {detected_errors}{RESET}")
    print(f"{GREEN}{'Success Rate':24}: {success_rate:.0f}%{RESET}")
    print(f"{YELLOW}{'Detected Error Rate':24}: {error_rate:.0f}%{RESET}")
    print("=" * 66)


if __name__ == "__main__":
    main()
